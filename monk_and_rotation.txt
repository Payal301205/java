import java.util.*;

class TestClass {
    public static void main(String[] args) throws Exception {
        Scanner sc = new Scanner(System.in);

        int t = sc.nextInt();

        while (t-- > 0) {
            int n = sc.nextInt();
            int k = sc.nextInt();

            int[] arr = new int[n];

            for (int i = 0; i < n; i++) {
                arr[i] = sc.nextInt();
            }

            k = k % n;  // Handle cases where k > n

            int[] ans = new int[n];
            int index = 0;

            // Copy last k elements
            for (int i = n - k; i < n; i++) {
                ans[index++] = arr[i];
            }

            // Copy remaining elements
            for (int i = 0; i < n - k; i++) {
                ans[index++] = arr[i];
            }

            // Print rotated array
            for (int i = 0; i < n; i++) {
                System.out.print(ans[i]);

                if (i < n - 1) {
                    System.out.print(" ");
                }
            }
            System.out.println();
        }

        sc.close();
    }
}
